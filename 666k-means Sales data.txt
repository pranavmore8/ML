
# 1. Import Required Libraries

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans, AgglomerativeClustering
from sklearn.decomposition import PCA

from scipy.cluster.hierarchy import dendrogram, linkage


# 2. Load Dataset

df = pd.read_csv("sales_data_sample.csv", encoding="ISO-8859-1")

print(df.head())
print(df.shape)
print(df.info())


# 3. Data Preprocessing

# Select numerical columns

X = df[
    [
        'QUANTITYORDERED',
        'PRICEEACH',
        'SALES',
        'MSRP'
    ]
]

# Check missing values

print(X.isnull().sum())

# Handle missing values

X = X.fillna(X.mean())

print(X.head())


# 4. Feature Scaling

scaler = StandardScaler()

X_scaled = scaler.fit_transform(X)

print("Scaled Data:")
print(X_scaled[:5])


# 5. Elbow Method

wcss = []

for i in range(1, 11):
    kmeans = KMeans(
        n_clusters=i,
        random_state=42,
        n_init=10
    )

    kmeans.fit(X_scaled)

    wcss.append(kmeans.inertia_)

# Plot Elbow Graph

plt.figure(figsize=(8, 5))

plt.plot(
    range(1, 11),
    wcss,
    marker='o'
)

plt.title("Elbow Method")
plt.xlabel("Number of Clusters")
plt.ylabel("WCSS")
plt.show()


# 6. Implement K-Means Clustering

kmeans = KMeans(
    n_clusters=3,
    random_state=42,
    n_init=10
)

df['KMeans_Cluster'] = kmeans.fit_predict(X_scaled)

print("\nK-Means Cluster Results:")
print(df['KMeans_Cluster'].value_counts())


# 7. Visualize K-Means Clusters using PCA

pca = PCA(n_components=2)

X_pca = pca.fit_transform(X_scaled)

plt.figure(figsize=(8, 6))

plt.scatter(
    X_pca[:, 0],
    X_pca[:, 1],
    c=df['KMeans_Cluster'],
    cmap='viridis'
)

plt.title("K-Means Clustering")
plt.xlabel("Principal Component 1")
plt.ylabel("Principal Component 2")
plt.show()


# 8. Hierarchical Clustering - Dendrogram

# Take a sample to reduce computation time

sample = X_scaled[:500]

plt.figure(figsize=(12, 6))

Z = linkage(sample, method='ward')

dendrogram(Z)

plt.title("Hierarchical Clustering Dendrogram")
plt.xlabel("Data Points")
plt.ylabel("Euclidean Distance")
plt.show()


# 9. Implement Hierarchical Clustering

hc = AgglomerativeClustering(
    n_clusters=3,
    linkage='ward'
)

df['Hierarchical_Cluster'] = hc.fit_predict(X_scaled)

print("\nHierarchical Clustering Results:")
print(df['Hierarchical_Cluster'].value_counts())


# 10. Visualize Hierarchical Clusters

plt.figure(figsize=(8, 6))

plt.scatter(
    X_pca[:, 0],
    X_pca[:, 1],
    c=df['Hierarchical_Cluster'],
    cmap='rainbow'
)

plt.title("Hierarchical Clustering")
plt.xlabel("Principal Component 1")
plt.ylabel("Principal Component 2")
plt.show()


# 11. Compare Cluster Results

print("\nK-Means Cluster Distribution:")
print(df['KMeans_Cluster'].value_counts())

print("\nHierarchical Cluster Distribution:")
print(df['Hierarchical_Cluster'].value_counts())


# 12. Cluster Analysis

print("\nK-Means Cluster Analysis:")

print(
    df.groupby('KMeans_Cluster')[
        ['QUANTITYORDERED', 'PRICEEACH', 'SALES', 'MSRP']
    ].mean()
)

print("\nHierarchical Cluster Analysis:")

print(
    df.groupby('Hierarchical_Cluster')[
        ['QUANTITYORDERED', 'PRICEEACH', 'SALES', 'MSRP']
    ].mean()
)
